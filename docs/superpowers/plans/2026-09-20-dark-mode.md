# Dark Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add OS-following dark mode with a per-account System/Light/Dark override, converting all screens/components from the static light palette to a themed provider.

**Architecture:** New `ThemeProvider` exposes `{ preference, scheme, colors, isDark, setPreference }`; light/dark palettes share identical semantic keys. Every themed file converts its module-level `StyleSheet.create` into a `makeStyles(colors)` factory consumed via `useThemedStyles`. Stored category color pairs are darkened/lightened at render time by a pure helper; a shared `CategoryAvatar` centralizes the icon-in-tinted-circle motif.

**Tech Stack:** Expo SDK 57, React Native 0.86, React 19, TypeScript strict, Jest 29 + jest-expo, React Navigation 7, `expo-system-ui` (new, SDK-bundled), Supabase.

**Spec:** `docs/superpowers/specs/2026-09-20-dark-mode-design.md`

**Implementation status:** Not started

## Recipe R — canonical style conversion (used by Phase 3 tasks)

Apply to one file at a time:

1. Change the theme import:
   - From: `import { colors, fonts, radii, shadows, text } from '../theme';`
   - To: `import { fonts, radii, shadows, text } from '../theme';` (drop `colors`, keep the rest that the file uses)
   - Add: `import { useTheme, useThemedStyles } from '../context/ThemeContext';`
   - Add a type import: `import type { ThemeColors } from '../theme';`
2. Rename `const styles = StyleSheet.create({ ... })` to `const makeStyles = (colors: ThemeColors) => StyleSheet.create({ ... })` — body unchanged.
3. Inside the component body (top, before any early return), add:
   ```tsx
   const styles = useThemedStyles(makeStyles);
   ```
4. If the component passes colors to props (e.g. `color={colors.primary}`, `placeholderTextColor`, `trackColor`, `ActivityIndicator color`), also add:
   ```tsx
   const { colors } = useTheme();
   ```
5. Components that render before/without a provider must not be converted; all UI components are under `ThemeProvider` after Task 4.
6. Do not convert data/test modules: `src/data/seed.ts`, `src/data/categoriesData.ts`, `src/data/repositories/mappers.ts`, `src/**/__tests__/**`.
7. After each file, run `npx tsc --noEmit` and fix remaining `colors` references (there should be none outside `makeStyles`/`useTheme`).

```ts
function parseHex(hex: string): [number, number, number] | null {
  const match = /^#?([0-9a-fA-F]{6})$/.exec(hex);
  if (!match) return null;
  const value = parseInt(match[1], 16);
  return [(value >> 16) & 255, (value >> 8) & 255, value & 255];
}

function channelToLinear(channel: number): number {
  const c = channel / 255;
  return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
}

export function relativeLuminance(hex: string): number | null {
  const rgb = parseHex(hex);
  if (!rgb) return null;
  return (
    0.2126 * channelToLinear(rgb[0]) +
    0.7152 * channelToLinear(rgb[1]) +
    0.0722 * channelToLinear(rgb[2])
  );
}

export function contrastRatio(a: string, b: string): number | null {
  const la = relativeLuminance(a);
  const lb = relativeLuminance(b);
  if (la === null || lb === null) return null;
  const lighter = Math.max(la, lb);
  const darker = Math.min(la, lb);
  return (lighter + 0.05) / (darker + 0.05);
}
```

```ts
interface Hsl {
  h: number;
  s: number;
  l: number;
}

const HEX = /^#?([0-9a-fA-F]{6})$/;
const FALLBACK_BG = '#1B2521';
const FALLBACK_FG = '#A3B0AB';

function hexToHsl(hex: string): Hsl | null {
  const match = HEX.exec(hex);
  if (!match) return null;
  const value = parseInt(match[1], 16);
  const r = ((value >> 16) & 255) / 255;
  const g = ((value >> 8) & 255) / 255;
  const b = (value & 255) / 255;
  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  const l = (max + min) / 2;
  let h = 0;
  let s = 0;
  if (max !== min) {
    const d = max - min;
    s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    if (max === r) h = (g - b) / d + (g < b ? 6 : 0);
    else if (max === g) h = (b - r) / d + 2;
    else h = (r - g) / d + 4;
    h *= 60;
  }
  return { h, s, l };
}

function hslToHex(h: number, s: number, l: number): string {
  const a = s * Math.min(l, 1 - l);
  const channel = (n: number): string => {
    const k = (n + h / 30) % 12;
    const c = l - a * Math.max(-1, Math.min(k - 3, 9 - k, 1));
    return Math.round(255 * c).toString(16).padStart(2, '0');
  };
  return `#${channel(0)}${channel(8)}${channel(4)}`.toUpperCase();
}

function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

export function deriveCategoryColors(
  colorBg: string,
  colorFg: string,
  isDark: boolean,
): { colorBg: string; colorFg: string } {
  if (!isDark) return { colorBg, colorFg };
  const bg = hexToHsl(colorBg);
  const fg = hexToHsl(colorFg);
  if (!bg || !fg) return { colorBg: FALLBACK_BG, colorFg: FALLBACK_FG };
  return {
    colorBg: hslToHex(bg.h, bg.s, clamp(bg.l, 0.22, 0.26)),
    colorFg: hslToHex(fg.h, fg.s, clamp(fg.l, 0.75, 0.82)),
  };
}
```

- [ ] **Step 1: Create the context**

```tsx
import React, { createContext, useContext, useEffect, useMemo, useState } from 'react';
import { useColorScheme } from 'react-native';
import * as SystemUI from 'expo-system-ui';
import { darkColors, lightColors, resolveScheme } from '../theme';
import type { ThemeColors, ThemePreference, ThemeScheme } from '../theme';

interface ThemeContextValue {
  preference: ThemePreference;
  scheme: ThemeScheme;
  colors: ThemeColors;
  isDark: boolean;
  setPreference: (preference: ThemePreference) => void;
}

const ThemeContext = createContext<ThemeContextValue | undefined>(undefined);

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const systemScheme = useColorScheme();
  const [preference, setPreference] = useState<ThemePreference>('system');
  const scheme = resolveScheme(preference, systemScheme);
  const colors = scheme === 'dark' ? darkColors : lightColors;

  useEffect(() => {
    SystemUI.setBackgroundColorAsync(colors.background).catch(() => {});
  }, [colors]);

  const value = useMemo(
    () => ({ preference, scheme, colors, isDark: scheme === 'dark', setPreference }),
    [preference, scheme, colors],
  );

  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>;
}

export function useTheme(): ThemeContextValue {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme must be used within ThemeProvider');
  return ctx;
}

export function useThemedStyles<T>(factory: (colors: ThemeColors) => T): T {
  const { colors } = useTheme();
  return useMemo(() => factory(colors), [factory, colors]);
}
```

```tsx
  const { user } = useAuth();

        setHapticsEnabled(existing?.hapticsEnabled ?? true);
        setCurrency(existing?.currency ?? 'USD ($)');
        setPreference(existing?.themePreference ?? 'system');
        if (!existing && user) {

          const name = (meta.full_name as string) || (meta.name as string) || user.email || 'You';
```
